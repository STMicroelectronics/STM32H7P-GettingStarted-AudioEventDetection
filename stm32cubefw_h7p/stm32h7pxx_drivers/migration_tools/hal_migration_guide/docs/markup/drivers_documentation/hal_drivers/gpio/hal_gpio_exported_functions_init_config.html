HAL GPIO initialization and configuration functions migration ¶

This section provides information about the HAL GPIO initialization and configuration functions migration.

GPIO pin configuration ¶

Create a new project with the STM32CubeMX2 , then start the initialisation and configuration.

GPIO pin selection ¶

  • Select PA0 as GPIO pin.

Configuration of the software layer and resource initialisation ¶

The HAL library is used for GPIO initialization, with resource initialization code set to be generated and called within the project.

GPIO mode and pull configuration ¶

  • Select GPIO pin mode as Input .

  • Select GPIO pin pull as No pull-upand no pull-down .

Generated GPIO initialization sequence ¶

The GPIO initialisation sequence is generated in the mx_gpio_default_hal_init function in mx_gpio_default.c file under $YOUR_PORJECT_NAME$/$YOUR_PORJECT_NAME$_SW/generated/STM32Cube_CodeGen.

void MX_GPIO_Init(void)
{

  GPIO_InitTypeDef GPIO_InitStruct = {0};

  /* GPIO Ports Clock Enable */
  __HAL_RCC_GPIOA_CLK_ENABLE();

  /*Configure GPIO pin : PA0 */
  GPIO_InitStruct.Pin = GPIO_PIN_0;
  GPIO_InitStruct.Mode = GPIO_MODE_INPUT;
  GPIO_InitStruct.Pull = GPIO_NOPULL;
  HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);

}

system_status_t mx_gpio_default_hal_init(void)
{
  hal_gpio_config_t  gpio_config;

  HAL_RCC_GPIOA_EnableClock();

  gpio_config.mode            = HAL_GPIO_MODE_INPUT;
  gpio_config.pull            = HAL_GPIO_PULL_NO;
  if (HAL_GPIO_Init(PA0_GPIO_PORT, PA0_PIN, &gpio_config) != HAL_OK)
  {
    return SYSTEM_PERIPHERAL_ERROR;
  }

#if defined(USE_HAL_GPIO_HSLV) && (USE_HAL_GPIO_HSLV == 1U)
  HAL_GPIO_DisableHighSpeedLowVoltage(PA0_GPIO_PORT, PA0_PIN);
#endif /* USE_HAL_GPIO_HSLV */

  return SYSTEM_OK;
}

void MX_GPIO_Init(void)
{

  GPIO_InitTypeDef GPIO_InitStruct = {0};

  /* GPIO Ports Clock Enable */
  __HAL_RCC_GPIOA_CLK_ENABLE();

  /*Configure GPIO pin : PA0 */
  GPIO_InitStruct.Pin = GPIO_PIN_0;
  GPIO_InitStruct.Mode = GPIO_MODE_IT_RISING;
  GPIO_InitStruct.Pull = GPIO_NOPULL;
  HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);

  /* EXTI interrupt init*/
  HAL_NVIC_SetPriority(EXTI0_IRQn, 0, 0);
  HAL_NVIC_EnableIRQ(EXTI0_IRQn);

}

system_status_t mx_gpio_default_hal_init(void)
{
  hal_gpio_config_t  gpio_config;

  HAL_RCC_GPIOA_EnableClock();

  gpio_config.mode            = HAL_GPIO_MODE_INPUT;
  gpio_config.pull            = HAL_GPIO_PULL_NO;
  if (HAL_GPIO_Init(HAL_GPIOA, HAL_GPIO_PIN_0, &gpio_config) != HAL_OK)
  {
    return SYSTEM_PERIPHERAL_ERROR;
  }

#if defined(USE_HAL_GPIO_HSLV) && (USE_HAL_GPIO_HSLV == 1U)
  HAL_GPIO_DisableHighSpeedLowVoltage(HAL_GPIOA, HAL_GPIO_PIN_0);
#endif /* USE_HAL_GPIO_HSLV */

  hal_exti_config_t exti_config;

  /* Initialize and configure the EXTI for GPIOA line0 */
  HAL_EXTI_Init(&hEXTI0, HAL_EXTI_LINE_0);

  exti_config.trigger   = HAL_EXTI_TRIGGER_RISING;
  exti_config.gpio_port = HAL_EXTI_GPIOA;

  HAL_EXTI_SetConfig(&hEXTI0, &exti_config);

  /* set line 0 Interrupt priority */
  HAL_CORTEX_NVIC_SetPriority(EXTI0_IRQn, HAL_CORTEX_NVIC_PREEMP_PRIORITY_0, HAL_CORTEX_NVIC_SUB_PRIORITY_0);
  HAL_CORTEX_NVIC_EnableIRQ(EXTI0_IRQn);

  return SYSTEM_OK;
}

GPIO deinit sequence. 

void MX_GPIO_DeInit(void)
{
  /* set line 0 Interrupt priority */
  HAL_NVIC_DisableIRQ(EXTI0_IRQn);

  HAL_GPIO_DeInit(GPIOA, GPIO_PIN_0);

}

system_status_t mx_gpio_default_hal_deinit(void)
{
  /* Deinitialize the EXTI for GPIOA line0 */
  HAL_EXTI_DeInit(&hEXTI0);

  /* set line 0 Interrupt priority */
  HAL_CORTEX_NVIC_DisableIRQ(EXTI0_IRQn);

  HAL_GPIO_DeInit(HAL_GPIOA, HAL_GPIO_PIN_0);

  return SYSTEM_OK;
}
