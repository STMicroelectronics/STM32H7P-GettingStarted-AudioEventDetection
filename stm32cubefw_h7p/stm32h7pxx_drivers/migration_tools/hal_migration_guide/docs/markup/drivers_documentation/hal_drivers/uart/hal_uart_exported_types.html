HAL UART states ¶

In HAL2, key changes include:

  1. Renaming of several states.

  2. Removal of states HAL_UART_STATE_TIMEOUT and HAL_UART_STATE_ERROR .

  3. Adding a new state HAL_UART_STATE_ABORT

  4. Refactoring of the state machine into three distinct state machines:

    • Global state machine: hal_uart_state_t

    • Reception state machine: hal_uart_rx_state_t

    • Transmission state machine: hal_uart_tx_state_t

HAL2

HAL UART advanced feature initialization structure ¶

HAL1

HAL2 

typedef struct
{
    uint32_t AdvFeatureInit;
    uint32_t TxPinLevelInvert;
    uint32_t RxPinLevelInvert;
    uint32_t DataInvert;
    uint32_t Swap;
    uint32_t OverrunDisable;
    uint32_t DMADisableonRxError;
    uint32_t AutoBaudRateEnable;
    uint32_t AutoBaudRateMode;
    uint32_t MSBFirst;
} UART_AdvFeatureInitTypeDef;

This structure has been removed, along with its association as a member of the handle.

Each advanced feature is now configured and controlled through specific functions, allowing direct management of these features without relying on the previous structure.

For more information on enabling or disabling advanced features, see the section HAL UART enable/disable advanced features for more details on advanced feature management functions.

Usage Sequence

HAL2 Usage 

UART_HandleTypeDef huart1;


// Add your basic config to the handle

huart1.AdvancedInit.AdvFeatureInit = UART_ADVFEATURE_TXINVERT_INIT |
                                     UART_ADVFEATURE_RXINVERT_INIT |
                                     UART_ADVFEATURE_DATAINVERT_INIT |
                                     UART_ADVFEATURE_SWAP_INIT |
                                     UART_ADVFEATURE_RXOVERRUNDISABLE_INIT |
                                     UART_ADVFEATURE_DMADISABLEONERROR_INIT |
                                     UART_ADVFEATURE_MSBFIRST_INIT |
                                     UART_ADVFEATURE_AUTOBAUDRATE_INIT;

huart1.AdvancedInit.TxPinLevelInvert = UART_ADVFEATURE_TXINV_ENABLE;
huart1.AdvancedInit.RxPinLevelInvert = UART_ADVFEATURE_RXINV_ENABLE;
huart1.AdvancedInit.DataInvert = UART_ADVFEATURE_DATAINV_ENABLE;
huart1.AdvancedInit.Swap = UART_ADVFEATURE_SWAP_ENABLE;
huart1.AdvancedInit.OverrunDisable = UART_ADVFEATURE_OVERRUN_DISABLE;
huart1.AdvancedInit.DMADisableonRxError = UART_ADVFEATURE_DMA_DISABLEONRXERROR;
huart1.AdvancedInit.AutoBaudRateEnable = UART_ADVFEATURE_AUTOBAUDRATE_ENABLE;
huart1.AdvancedInit.AutoBaudRateMode = UART_ADVFEATURE_AUTOBAUDRATE_ONSTARTBIT;
huart1.AdvancedInit.MSBFirst = UART_ADVFEATURE_MSBFIRST_ENABLE;

/* Initialization and configure basic and advanced configuration features selected  */
HAL_UART_Init(&huart1);

hal_uart_handle_t hUART1;
hal_uart_config_t uart_config;

// Add your basic config

/* Initialization */
HAL_UART_Init(&hUART1,HAL_UART1);

/* Basic configuration */
HAL_UART_SetConfig(&hUART1, &uart_config);



/* Configure and enable advanced features */
HAL_UART_EnableTxPinLevelInvert(&hUART1);
HAL_UART_EnableRxPinLevelInvert(&hUART1);

HAL_UART_DisableRxOverRunDetection(&hUART1);
HAL_UART_EnableDMAStopOnRxError(&hUART1);
HAL_UART_SetConfigAutoBaudRateMode(&hUART1, HAL_UART_AUTO_BAUD_DET_ON_START_BIT);

HAL_UART_EnableMSBFirst(&hUART1);
