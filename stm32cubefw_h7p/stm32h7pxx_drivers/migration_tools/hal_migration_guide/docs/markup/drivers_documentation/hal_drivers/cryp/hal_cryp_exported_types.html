This section provides information about the HAL CRYP types migration .

Init structure definition ¶

In HAL2, each algorithm has its own function configuration, so we no longer require a global configuration structure.

HAL1

HAL2 

typedef struct
{
  uint32_t *pInitVect;
  uint32_t *Header;
  uint32_t HeaderSize;
  uint32_t DataType;
  uint32_t KeySize;
  uint32_t *pKey;
  uint32_t *B0;
  uint32_t DataWidthUnit;
  uint32_t HeaderWidthUnit;
  uint32_t KeyIVConfigSkip;


  uint32_t Algorithm;

  uint32_t KeyMode;
  uint32_t KeySelect;
  uint32_t KeyProtection;
} CRYP_ConfigTypeDef, CRYP_InitTypeDef;

typedef struct
{
  uint32_t *p_init_vect;
  uint32_t *p_header;
  uint32_t header_size_byte ;
} hal_cryp_gcm_config_t;

typedef struct
{
  uint32_t *p_b0;
  uint32_t *p_header;
  uint32_t header_size_byte;
} hal_cryp_ccm_config_t;

New Configuration functions for each algorithm, example HAL_CRYP_AES_ECB_SetConfig().


New set key functions are used to configure the key example HAL_CRYP_AES_SetNormalKey().

Related concepts:

  • Rename unitary init structures

  • Replace the “TypeDef” part of every typedef with “_t”.

Context structure definition ¶

The contents of the Context structure are internal to the driver and do not need to be modified by the user.

HAL1

HAL2 

typedef struct
{
  uint32_t CR_Reg;
  uint32_t IVR0_Reg;
  uint32_t IVR1_Reg;
  uint32_t IVR2_Reg;
  uint32_t IVR3_Reg;

  uint32_t DataType;
  uint32_t KeySize;
  uint32_t *pKey;
  uint32_t *pInitVect;
  uint32_t Algorithm;
  uint32_t DataWidthUnit;
  uint32_t KeyIVConfigSkip;
  uint32_t KeyMode;
  uint32_t Phase;
  uint32_t KeyIVConfig;
  uint32_t IER_Reg;
} CRYP_ContextTypeDef;

typedef struct
{
  uint32_t CR;
  uint32_t iv_buff[4];
  uint32_t SUSPxR[8];
  hal_cryp_t                 instance;
  volatile  hal_cryp_state_t previous_state;
  uint32_t                  algorithm;
  volatile uint32_t         data_size_byte;
  volatile uint32_t         data_size_sum_byte;
  const uint32_t            *p_in_buff;
  uint32_t                  *p_out_buff;
  volatile uint32_t         block_count;
  const uint32_t            *p_header;
  uint32_t                  header_size_byte ;
  volatile uint32_t         suspend_request;
  const uint32_t            *p_key;
  hal_cryp_cb_t               p_in_cplt_cb;
  hal_cryp_cb_t               p_out_cplt_cb;
  hal_cryp_cb_t               p_error_cb;
  hal_cryp_cb_t               p_suspend_cb;

} hal_cryp_save_context_t;
