LL_FMAC_Init and LL_FMAC_DeInit APIs migration ¶

The LL initialization and de-initialization functions previously available in the stm32tnxx_ll_fmac.c files within HAL1 are no longer present in HAL2. In HAL2, all LL C files ( stm32tnxx_ll_fmac.c ) have been removed, leaving only the LL header files ( stm32tnxx_ll_fmac.h ) that contain static inline functions.

If users need to initialize or de-initialize FMAC using the LL layer, they have two methods:

  • Utilize STM32CubeMX2 Code Generation : Rely on STM32CubeMX2 to generate code with the LL layer for FMAC initialization and de-initialization.

  • Replace manually the initialization and de-initialization functions Replace the LL_FMAC_Init and LL_FMAC_DeInit functions

    manually with equivalent sequences using the static inline functions provided in the stm32tnxx_ll_fmac.h file, tailored to their specific configuration and requirements.

This approach ensures that the initialization and de-initialization sequences are precisely addressing the given use case based on the LL static inline functions provided in the stm32tnxx_ll_fmac.h header file. The sequences offer improved footprint optimization compared to the previous LL initialization functions found in stm32tnxx_ll_fmac.c .

FMAC_TypeDef *mx_fmac_ll_init(void)
{
  LL_EnableClock(LL_PERIPH_FMAC);

  LL_FMAC_SetX1BaseAndSize(FMAC, 2, 12);
  LL_FMAC_SetX2BaseAndSize(FMAC, 15, 12);
  LL_FMAC_SetYBaseAndSize(FMAC, 27, 12);

  /* Configure the DMA Write */

  /* ################################################## FMAC_wr DMA configuration */
  /* Enable GPDMA1 clock */
  LL_AHB1_GRP1_EnableClock(LL_AHB1_GRP1_PERIPH_GPDMA1);

  /* Configure DMA channel transfer request */
  LL_DMA_SetPeriphRequest(GPDMA1_CH0, LL_GPDMA1_REQUEST_FMAC_WR);

  /* Configure DMA channel transfer direction */
  LL_DMA_SetDataTransferDirection(GPDMA1_CH0, LL_DMA_DIRECTION_MEMORY_TO_MEMORY);

  /* Configure DMA channel priority level */
  LL_DMA_SetChannelPriorityLevel(GPDMA1_CH0, LL_DMA_PRIORITY_LOW_WEIGHT_LOW);

  /* Configure DMA channel transfer register 1 */
  LL_DMA_ConfigTransfer(GPDMA1_CH0,
                        (LL_DMA_SRC_ADDR_FIXED | LL_DMA_SRC_DATA_WIDTH_BYTE |
                         LL_DMA_DEST_ADDR_FIXED | LL_DMA_DEST_DATA_WIDTH_BYTE));

  /* Configure the DMA Read */

  /* ################################################## FMAC_rd DMA configuration */
  /* Enable GPDMA1 clock */
  LL_AHB1_GRP1_EnableClock(LL_AHB1_GRP1_PERIPH_GPDMA1);

  /* Configure DMA channel transfer request */
  LL_DMA_SetPeriphRequest(GPDMA1_CH1, LL_GPDMA1_REQUEST_FMAC_RD);

  /* Configure DMA channel transfer direction */
  LL_DMA_SetDataTransferDirection(GPDMA1_CH1, LL_DMA_DIRECTION_MEMORY_TO_MEMORY);

  /* Configure DMA channel priority level */
  LL_DMA_SetChannelPriorityLevel(GPDMA1_CH1, LL_DMA_PRIORITY_LOW_WEIGHT_LOW);

  /* Configure DMA channel transfer register 1 */
  LL_DMA_ConfigTransfer(GPDMA1_CH1,
                        (LL_DMA_SRC_ADDR_FIXED | LL_DMA_SRC_DATA_WIDTH_BYTE |
                         LL_DMA_DEST_ADDR_FIXED | LL_DMA_DEST_DATA_WIDTH_BYTE));

  /* Enable the Event interruption for FMAC */
  NVIC_SetPriority(FMAC_IRQn, NVIC_EncodePriority(NVIC_GetPriorityGrouping(), 0, 0));
  NVIC_EnableIRQ(FMAC_IRQn);

  return FMAC;
}

void mx_fmac_ll_deinit(void)
{
  /* Disable the Event interruption for FMAC */
  NVIC_DisableIRQ(FMAC_IRQn);
  LL_FMAC_EnableReset(FMAC);

  /* Disable the DMA Write */
  /* De-initialize the DMA channel */
  LL_DMA_DisableChannel(GPDMA1_CH0);
  LL_DMA_ClearFlag(GPDMA1_CH0, LL_DMA_FLAG_ALL);

  LL_DMA_WRITE_REG(GPDMA1_CH0, CCR, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH0, CTR1, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH0, CTR2, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH0, CBR1, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH0, CTR3, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH0, CBR2, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH0, CLLR, 0U);

  /* Disable the DMA Read */
  /* De-initialize the DMA channel */
  LL_DMA_DisableChannel(GPDMA1_CH1);
  LL_DMA_ClearFlag(GPDMA1_CH1, LL_DMA_FLAG_ALL);

  LL_DMA_WRITE_REG(GPDMA1_CH1, CCR, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH1, CTR1, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH1, CTR2, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH1, CBR1, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH1, CTR3, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH1, CBR2, 0U);
  LL_DMA_WRITE_REG(GPDMA1_CH1, CLLR, 0U);
}
