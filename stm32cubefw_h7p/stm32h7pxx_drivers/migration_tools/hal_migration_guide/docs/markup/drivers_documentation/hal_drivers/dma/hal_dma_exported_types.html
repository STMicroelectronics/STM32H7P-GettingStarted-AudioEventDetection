Note

The callback registration in HAL2 is handled by specific functions for each DMA event. These functions are used to register callback functions for different DMA events. See Callbacks register Callbacks functions for more details.

HAL DMA data handling configuration ¶

In HAL2, DataExchange is replaced with hal_dma_src_byte_exchange_t , hal_dma_dest_byte_exchange_t , and hal_dma_dest_halfword_exchange_t . DataAlignment is replaced with hal_dma_dest_data_trunc_padd_t and hal_dma_dest_data_pack_t .

HAL1

HAL2 

typedef struct
{
  uint32_t DataExchange;



  uint32_t DataAlignment;

} DMA_DataHandlingConfTypeDef;

typedef struct
{
  hal_dma_src_byte_exchange_t      src_byte_exchange;
  hal_dma_dest_byte_exchange_t     dest_byte_exchange;
  hal_dma_dest_halfword_exchange_t dest_halfword_exchange;

  hal_dma_dest_data_trunc_padd_t   trunc_padd;
  hal_dma_dest_data_pack_t         pack;
} hal_dma_data_handling_config_t;

HAL DMA Linked-List Queue ¶

In HAL2, DMA_QListTypeDef structure is removed from HAL DMA driver and replaced with hal_q_t in the HAL Q driver .

HAL1

HAL2 

typedef struct
{
  DMA_NodeTypeDef            *Head;

  DMA_NodeTypeDef            *FirstCircularNode;
  uint32_t                   NodeNumber;

  __IO HAL_DMA_QStateTypeDef State;
  __IO uint32_t              ErrorCode;

  __IO uint32_t              Type;





} DMA_QListTypeDef;

typedef struct
{
    void     *p_head_node;
    void     *p_tail_node;
    void     *p_first_circular_node;
    uint32_t node_nbr;

    There is no state machine for the HAL QUEUE driver, so the queue state is not needed.
    The HAL QUEUE error code is not related to the physical instance and is not needed for the user.

    In HAL2, the queue type is not needed for the user.

    uint32_t next_addr_offset;
    hal_q_addressing_mode_t addr_mode;
    void (* p_set_node)(uint32_t head, uint32_t prev, uint32_t next, uint32_t offset);
    uint32_t (* p_get_node)(uint32_t head, uint32_t node, uint32_t offset);
} hal_q_t;
