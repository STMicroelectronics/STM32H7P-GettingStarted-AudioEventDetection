HAL RNG initialization and configuration functions migration ¶

The goal of this page is to show how to migrate the Init sequence using STM32CubeMX2:

  • Open STM32CubeMX2 RNG configuration panel

  • Configure the RNG

  • Configure the SW layer HAL/LL

  • Configure the RNG compliant

  • Configure the additional features of the RNG if needed like Clock error detection

  • Configure the NVIC if the application requires the RNG interrupts

  • Generate the code

Note

The configuration below might varies from a series to another.

Generated RNG initialization sequence ¶

  • Generate the code

  • The RNG initialisation sequence is generated in the mx_rng_default_hal_init function in mx_rng.c file under $YOUR_PORJECT_NAME$/$YOUR_PORJECT_NAME$_SW/generated/STM32Cube_CodeGen .

  • No Msp Callback in HAL2.

Related concept :

  • Split legacy HAL_PPP_Init into Init (build handle object) and config (apply configuration)

  • Remove MspInit and MspDeInit callbacks

Init, config and Msp sequence function 

static void MX_RNG_Init(void)
{

  hrng.Instance = RNG;
  hrng.Init.ClockErrorDetection = RNG_CED_ENABLE;
  if (HAL_RNG_Init(&hrng) != HAL_OK)
  {
    Error_Handler();
  }

}












void HAL_RNG_MspInit(RNG_HandleTypeDef* hrng)
{
  RCC_PeriphCLKInitTypeDef PeriphClkInit = {0};
  if(hrng->Instance==RNG)
  {

    PeriphClkInit.PeriphClockSelection = RCC_PERIPHCLK_RNG;
    PeriphClkInit.RngClockSelection = RCC_RNGCLKSOURCE_HSI48;
    if (HAL_RCCEx_PeriphCLKConfig(&PeriphClkInit) != HAL_OK)
    {
      Error_Handler();
    }

    /* Peripheral clock enable */
    __HAL_RCC_RNG_CLK_ENABLE();
  }

}

hal_rng_handle_t *mx_rng_hal_init(void)
{
    HAL_RCC_RNG_EnableClock();

    if (HAL_RNG_Init(&hRNG, HAL_RNG) != HAL_OK)
    {
        return NULL;
    }

    /* Code HAL_RNG_Config....*/
    HAL_RNG_SetCandidateNISTConfig(&hRNG);
    if (HAL_RCC_RNG_SetKernelClkSource(HAL_RCC_RNG_CLK_SRC_HSI48) != HAL_OK)
    {
        return NULL;
    }

    HAL_RNG_EnableClockErrorDetection(&hRNG);

    /* Enable the interruption for RNG */

    return &hRNG;
}

Deinit sequence function 

void MX_RNG_DeInit(void)
{
  if (HAL_RNG_DeInit(&hrng) != HAL_OK)
  {
    Error_Handler();
  }
}





void HAL_RNG_MspDeInit(RNG_HandleTypeDef* hrng)
{
  if(hrng->Instance==RNG)
  {
    __HAL_RCC_RNG_CLK_DISABLE();

    __HAL_RCC_RNG_FORCE_RESET();

    __HAL_RCC_RNG_RELEASE_RESET();

  }

}

void mx_rng_hal_deinit(void)
{
    /* Disable the interruption for RNG */

    (void)HAL_RNG_DeInit(&hRNG);

    HAL_RCC_RNG_Reset();

    HAL_RCC_RNG_DisableClock();
}
