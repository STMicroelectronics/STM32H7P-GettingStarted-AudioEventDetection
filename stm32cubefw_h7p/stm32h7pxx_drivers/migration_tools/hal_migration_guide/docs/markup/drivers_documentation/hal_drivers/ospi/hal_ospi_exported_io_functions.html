Example usage ¶

HAL1

HAL2 

OSPI_HandleTypeDef hospi1;
OSPIM_CfgTypeDef sOspiManagerCfg;

/* Configure port 1 */
sOspiManagerCfg.DQSPort = 1;
sOspiManagerCfg.NCSPort = 1;
sOspiManagerCfg.IOPort = HAL_OSPIM_IOPORT_1_LOW;
HAL_OSPIM_Config(&hospi1, &sOspiManagerCfg, HAL_XSPI_TIMEOUT_DEFAULT_VALUE);


/* Configure port 2 */
sOspiManagerCfg.DQSPort = 1;
sOspiManagerCfg.NCSPort = 1;
sOspiManagerCfg.IOPort = HAL_XSPIM_IOPORT_2_LOW;
HAL_OSPIM_Config(&hospi1, &sOspiManagerCfg, HAL_XSPI_TIMEOUT_DEFAULT_VALUE);

hal_xspi_xspim_config_t sOspiManagerCfg;


/* Configure the OCTOSPIM port 1 */
sOspiManagerCfg.port1.clk = HAL_XSPI_XSPIM_XSPI1;
sOspiManagerCfg.port1.dqs = HAL_XSPI_XSPIM_XSPI1;
sOspiManagerCfg.port1.ncs = HAL_XSPI_XSPIM_XSPI1;
sOspiManagerCfg.port1.io_low = HAL_XSPI_XSPIM_XSPI1_IO_LOW;
sOspiManagerCfg.port1.io_high = HAL_XSPI_XSPIM_XSPI1_IO_HIGH

/* Configure the OCTOSPIM port 2 */
sOspiManagerCfg.port2.clk = HAL_XSPI_XSPIM_XSPI2;
sOspiManagerCfg.port2.dqs = HAL_XSPI_XSPIM_XSPI2;
sOspiManagerCfg.port2.ncs = HAL_XSPI_XSPIM_XSPI2;
sOspiManagerCfg.port2.io_low = HAL_XSPI_XSPIM_XSPI2_IO_LOW;
sOspiManagerCfg.port2.io_high = HAL_XSPI_XSPIM_XSPI2_IO_HIGH;

HAL_XSPI_XSPIM_SetConfig(&sOspiManagerCfg);
