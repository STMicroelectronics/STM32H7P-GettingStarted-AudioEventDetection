DAC using software trigger mode ¶

HAL1

HAL2

Sequence diagram

Usage 

DAC_ChannelConfTypeDef sConfig;
HAL_DAC_Init(&hdac1);
sConfig.DAC_HighFrequency = DAC_HIGH_FREQUENCY_INTERFACE_MODE_ABOVE_80MH;
sConfig.DAC_DMADoubleDataMode = DISABLE;
sConfig.DAC_SignedFormat = DISABLE;
sConfig.DAC_SampleAndHold = DAC_SAMPLEANDHOLD_DISABLE;
sConfig.DAC_Trigger = DAC_TRIGGER_SOFTWARE;
sConfig.DAC_OutputBuffer = DAC_OUTPUTBUFFER_ENABLE;
sConfig.DAC_ConnectOnChipPeripheral = DAC_CHIPCONNECT_EXTERNAL;
sConfig.DAC_UserTrimming = DAC_TRIMMING_FACTORY;
sConfig.DAC_TrimmingValue = 31;
HAL_DAC_ConfigChannel(&hdac1, &sConfig, DAC_CHANNEL_1);

/* The calibration of DAC is Optional */
HAL_DACEx_SelfCalibrate(&hdac1, &sConfig, DAC_CHANNEL_1);
HAL_DACEx_SetUserTrimming(&hdac1, &sConfig, DAC_CHANNEL_1, 31);

HAL_DAC_SetValue(&hdac, DAC_CHANNEL_1,
                 DAC_ALIGN_12B_R, 2048);
HAL_DAC_Start(&hdac, DAC_CHANNEL_1);
HAL_DAC_Stop(&hdac, DAC_CHANNEL_1);

hal_dac_config_t p_config;
hal_dac_channel_config_t p_config1;

HAL_DAC_Init(&hdac, HAL_DAC1);
p_config.high_frequency_mode = HAL_DAC_HIGH_FREQ_MODE_ABOVE_80MHZ;
HAL_DAC_SetConfig(&hdac, &p_config);

p_config1.data_sign_format = HAL_DAC_SIGN_FORMAT_UNSIGNED;
p_config1.data_trigger = HAL_DAC_TRIGGER_SOFTWARE;
p_config1.output_buffer = HAL_DAC_OUTPUT_BUFFER_ENABLED;
p_config1.output_connection = HAL_DAC_OUTPUT_CONNECTION_EXTERNAL;
p_config1.alignment = HAL_DAC_DATA_ALIGN_12_BITS_RIGHT;

HAL_DAC_SetConfigChannel(&hdac, HAL_DAC_CHANNEL_1, &p_config1);

/* The calibration of DAC is Optional */
HAL_DAC_CalibrateChannelBuffer(&hdac, HAL_DAC_CHANNEL_1);
HAL_DAC_SetChannelBufferCalibrationValue(&hdac, HAL_DAC_CHANNEL_1, 31);

HAL_DAC_SetChannelAlignment(&hdac, HAL_DAC_CHANNEL_1,
                            HAL_DAC_DATA_ALIGN_12_BITS_RIGHT);
HAL_DAC_SetChannelData(&hdac, HAL_DAC_CHANNEL_1,
                       2048);
HAL_DAC_StartChannel(&hdac, HAL_DAC_CHANNEL_1);
HAL_DAC_TrigSWConversionChannel(&hdac, HAL_DAC_CHANNEL_1);
HAL_DAC_StopChannel(&hdac, HAL_DAC_CHANNEL_);

Single Channel DAC Transfer Using DMA ¶

HAL1

HAL2

Usage 

uint32_t aEscalator12bit[6] = {0x0, 0x330, 0x660, 0x990, 0xCC0, 0xFF0};
DAC_ChannelConfTypeDef sConfig;
sConfig.DAC_HighFrequency = DAC_HIGH_FREQUENCY_INTERFACE_MODE_ABOVE_80MH;
sConfig.DAC_DMADoubleDataMode = DISABLE;
sConfig.DAC_SignedFormat = DISABLE;
sConfig.DAC_SampleAndHold = DAC_SAMPLEANDHOLD_DISABLE;
sConfig.DAC_Trigger = DAC_TRIGGER_SOFTWARE;
sConfig.DAC_OutputBuffer = DAC_OUTPUTBUFFER_ENABLE;
sConfig.DAC_ConnectOnChipPeripheral = DAC_CHIPCONNECT_EXTERNAL;
sConfig.DAC_UserTrimming = DAC_TRIMMING_FACTORY;
sConfig.DAC_TrimmingValue = 31;
HAL_DAC_ConfigChannel(&hdac1, &sConfig, DAC_CHANNEL_1);

HAL_DAC_SetValue(&hdac1, DAC_CHANNEL_1, DAC_ALIGN_12B_R, 2048);

HAL_DAC_Start_DMA(&hdac1, DAC_CHANNEL_1, aEscalator8bit, 6, DAC_ALIGN_12B_R);

HAL_DAC_Stop_DMA(&hdac1, DAC_CHANNEL_1);

uint32_t aEscalator12bit[6] = {0x0, 0x330, 0x660, 0x990, 0xCC0, 0xFF0};
hal_dac_config_t p_config;
hal_dac_channel_config_t p_config1;

p_config.high_frequency_mode = HAL_DAC_HIGH_FREQ_MODE_ABOVE_80MHZ;
HAL_DAC_SetConfig(&hdac, &p_config);

p_config1.data_sign_format = HAL_DAC_SIGN_FORMAT_UNSIGNED;
p_config1.data_trigger = HAL_DAC_TRIGGER_TIM1_TRGO;
p_config1.output_buffer = HAL_DAC_OUTPUT_BUFFER_ENABLED;
p_config1.output_connection = HAL_DAC_OUTPUT_CONNECTION_EXTERNAL;
p_config1.alignment = HAL_DAC_DATA_ALIGN_12_BITS_RIGHT;
HAL_DAC_SetConfigChannel(&hdac, HAL_DAC_CHANNEL_1,
                         &p_config1);

HAL_DAC_SetChannelAlignment(&hdac, HAL_DAC_CHANNEL_1,
                             HAL_DAC_DATA_ALIGN_12_BITS_RIGHT);
HAL_DAC_SetChannelData(&hdac, HAL_DAC_CHANNEL_1,
                         2048);

HAL_DAC_StartChannel_DMA(&hdac, HAL_DAC_CHANNEL_1, aEscalator8bit, 6);

HAL_DAC_StopChannel_DMA(&hdac, HAL_DAC_CHANNEL_1);

Dual Channel DAC Transfer Using DMA ¶

HAL1

HAL2

Usage 

uint32_t aEscalator12bit[6] = {0x0, 0x330, 0x660, 0x990, 0xCC0, 0xFF0};
DAC_ChannelConfTypeDef sConfig;

sConfig.DAC_HighFrequency = DAC_HIGH_FREQUENCY_INTERFACE_MODE_ABOVE_80MH;
sConfig.DAC_DMADoubleDataMode = DISABLE;
sConfig.DAC_SignedFormat = DISABLE;
sConfig.DAC_SampleAndHold = DAC_SAMPLEANDHOLD_DISABLE;
sConfig.DAC_Trigger = DAC_TRIGGER_T1_TRGO;
sConfig.DAC_OutputBuffer = DAC_OUTPUTBUFFER_ENABLE;
sConfig.DAC_ConnectOnChipPeripheral = DAC_CHIPCONNECT_EXTERNAL;
sConfig.DAC_UserTrimming = DAC_TRIMMING_FACTORY;
sConfig.DAC_TrimmingValue = 31;
HAL_DAC_ConfigChannel(&hdac1, &sConfig, DAC_CHANNEL_1);
HAL_DAC_ConfigChannel(&hdac1, &sConfig, DAC_CHANNEL_2);

HAL_DACEx_DualSetValue(&hdac1, DAC_ALIGN_12B_R, 0x0FFF , 0x07FF);

HAL_DACEx_DualStart_DMA(&hdac1, DAC_CHANNEL_1, aEscalator12bit, 6, DAC_ALIGN_12B_R);

HAL_DACEx_DualStop_DMA(&hdac1, DAC_CHANNEL_1);

uint32_t aEscalator12bit[6] = {0x0, 0x330, 0x660, 0x990, 0xCC0, 0xFF0};
hal_dac_dual_channel_config_t dacDualChannelConfig;

hal_dac_config_t p_config;
p_config.high_frequency_mode = HAL_DAC_HIGH_FREQ_MODE_ABOVE_80MHZ;
HAL_DAC_SetConfig(&hdac, &p_config);
dacDualChannelConfig.channel1_config_t.data_sign_format  = HAL_DAC_SIGN_FORMAT_UNSIGNED;
dacDualChannelConfig.channel1_config_t.trigger           = HAL_DAC_TRIGGER_TIM1_TRGO;
dacDualChannelConfig.channel1_config_t.output_buffer     = HAL_DAC_OUTPUT_BUFFER_ENABLED;
dacDualChannelConfig.channel1_config_t.output_connection = HAL_DAC_OUTPUT_CONNECTION_EXTERNAL;

/* Config Channel 2 */
dacDualChannelConfig.channel2_config_t.data_sign_format  = HAL_DAC_SIGN_FORMAT_UNSIGNED;
dacDualChannelConfig.channel2_config_t.trigger           = HAL_DAC_TRIGGER_TIM1_TRGO;
dacDualChannelConfig.channel2_config_t.output_buffer     = HAL_DAC_OUTPUT_BUFFER_ENABLED;
dacDualChannelConfig.channel2_config_t.output_connection = HAL_DAC_OUTPUT_CONNECTION_EXTERNAL;

dacDualChannelConfig.alignment = HAL_DAC_DATA_ALIGN_12_BITS_RIGHT;

HAL_DAC_SetConfigDualChannel(&hdac, &dacDualChannelConfig);
HAL_DAC_SetDualChannelData(&hdac, 0x0FFF);

HAL_DAC_StartDualChannel_DMA(&hdac, aEscalator12bit, 6);

HAL_DAC_StopDualChannel_DMA(&hdac);
