HAL UART typical use cases for migration ¶

Note

The processes for transmitting and receiving data using polling, interrupt or DMA are the same in HAL1 and HAL2, except for the specification of buffer size: HAL1 uses Size for the number of data elements (u8 or u16), while HAL2 uses Size_byte for the buffer size in bytes.

#define DATA_TO_SEND_SIZE           256
#define DATA_TO_RECEIVE_SIZE       DATA_TO_SEND_SIZE

uint8_t aTxBuffer[] = " **** UART Communication **** ";
uint8_t aRxBuffer[DATA_TO_RECEIVE_SIZE];

UART_HandleTypeDef huart;


HAL_UART_Init(&huart)




HAL_UART_Receive(&huart, aRxBuffer,
                      DATA_TO_RECEIVE_SIZE, 100)


HAL_UART_Transmit(&huart, aTxBuffer,
                      DATA_TO_SEND_SIZE, 100)

#define DATA_TO_SEND_SIZE           256
#define DATA_TO_RECEIVE_SIZE       DATA_TO_SEND_SIZE

uint8_t aTxBuffer[] = " **** UART Communication **** ";
uint8_t aRxBuffer[DATA_TO_RECEIVE_SIZE];

hal_uart_handle_t huart;
hal_uart_config_t uart_config;

HAL_UART_Init(&huart, HAL_UART4)

/* add config for uart instance */
HAL_UART_SetConfig(&huart, &uart_config)

HAL_UART_Receive(&huart, aRxBuffer,
                      DATA_TO_RECEIVE_SIZE, 100)


HAL_UART_Transmit(&huart,aTxBuffer,
                      DATA_TO_SEND_SIZE, 100)
