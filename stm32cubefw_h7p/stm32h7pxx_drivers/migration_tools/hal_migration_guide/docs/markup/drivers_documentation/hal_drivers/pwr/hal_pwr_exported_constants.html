HAL PWR constants migration ¶

This section provides information about the HAL PWR constants migration .

HAL PWR Memory retention ¶

HAL1

HAL2

  • HAL1 uses separate defines for each page given to the function HAL_PWREx_EnableRAMsContentStopRetention

  • HAL2 use one single define : HAL_PWR_MEMORY_RETENTION_SRAM6_STOP0123, and a function HAL_PWR_LP_EnableMemoryPageRetention taking as parameters this define plus the first page and the number of pages 

#define PWR_SRAM6_PAGE8_STOP
#define PWR_SRAM6_PAGE7_STOP
#define PWR_SRAM6_PAGE6_STOP
#define PWR_SRAM6_PAGE5_STOP
#define PWR_SRAM6_PAGE4_STOP
#define PWR_SRAM6_PAGE3_STOP
#define PWR_SRAM6_PAGE2_STOP
#define PWR_SRAM6_PAGE1_STOP
#define PWR_SRAM6_FULL_STOP

#define HAL_PWR_MEMORY_RETENTION_SRAM6_STOP0123
/* and */
/* For PAGEx you need to specify the page_indx and page_nbr */

Example 

HAL_PWREx_EnableRAMsContentStopRetention(PWR_SRAM6_PAGEx_STOP)

Example 

HAL_PWR_LP_EnableMemoryPageRetention(HAL_PWR_MEMORY_RETENTION_SRAM6_STOP0123 ,
                                     page_idx,
                                     page_nbr)

HAL pwr trigger mode constants ¶

HAL1

HAL2

  • The PWR trigger mode is managed using defines.

  • The EXTI mode and trigger are defined separately using distinct enumerations hal_exti_mode_t and hal_exti_trigger_t in HAL EXTI driver. 

/* */

#define  PWR_PVM_MODE_NORMAL
#define  PWR_AVD_MODE_NORMAL
#define  PWR_PVD_MODE_NORMAL

#define  PWR_AVD_MODE_IT_RISING_FALLING
#define  PWR_AVD_MODE_IT_FALLING
#define  PWR_AVD_MODE_IT_RISING

#define  PWR_PVD_MODE_IT_RISING_FALLING

#define  PWR_PVD_MODE_IT_RISING

#define  PWR_PVM_MODE_EVENT_RISING
#define  PWR_PVM_MODE_EVENT_FALLING
#define  PWR_PVM_MODE_EVENT_RISING_FALLING

#define  PWR_AVD_MODE_EVENT_RISING
#define  PWR_AVD_MODE_EVENT_FALLING
#define  PWR_AVD_MODE_EVENT_RISING_FALLING

#define  PWR_PVD_MODE_EVENT_RISING

#define  PWR_PVD_MODE_EVENT_RISING_FALLING

 /*! EXTI Triggers */
 typedef enum
 {
   HAL_EXTI_TRIGGER_NONE
   HAL_EXTI_TRIGGER_RISING

   HAL_EXTI_TRIGGER_FALLING

   HAL_EXTI_TRIGGER_RISING_FALLING

 } hal_exti_trigger_t;

 /*! EXTI Modes */
 typedef enum
 {
   HAL_EXTI_MODE_INTERRUPT
   HAL_EXTI_MODE_EVENT
   HAL_EXTI_MODE_INTERRUPT_EVENT
 } hal_exti_mode_t;

/*Example usage*/

EXTI_ConfigStructure.trigger = HAL_EXTI_TRIGGER_RISING_FALLING;
EXTI_ConfigStructure.gpio_port = HAL_EXTI_GPIOA;
HAL_EXTI_Init(&hexti, HAL_EXTI_LINE_16);
HAL_EXTI_SetConfig(&hexti, &EXTI_ConfigStructure);
HAL_EXTI_Enable(&hexti, HAL_EXTI_MODE_INTERRUPT);
