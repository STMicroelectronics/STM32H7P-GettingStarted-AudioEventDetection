HAL TIM types migration ¶

This section provides information about the HAL TIM types migration .

 */

} HAL_TIM_ChannelStateTypeDef;

/**
 * @brief  TIM Channel States definition
 */
typedef enum
{
  HAL_TIM_CHANNEL_STATE_RESET

  HAL_TIM_OC_CHANNEL_STATE_IDLE
  HAL_TIM_IC_CHANNEL_STATE_IDLE

  HAL_TIM_OC_CHANNEL_STATE_ACTIVE
  HAL_TIM_IC_CHANNEL_STATE_ACTIVE

  #if defined(USE_HAL_TIM_DMA) && (USE_HAL_TIM_DMA == 1)
  /** An internal process is ongoing on the TIM output channel in DMA silent mode */
  HAL_TIM_OC_CHANNEL_STATE_ACTIVE_SILENT
  #endif /* USE_HAL_TIM_DMA */

  #if defined(USE_HAL_TIM_DMA) && (USE_HAL_TIM_DMA == 1)
  /** An internal process is ongoing on the TIM input channel in DMA silent mode */
  HAL_TIM_IC_CHANNEL_STATE_ACTIVE_SILENT   = (HAL_TIM_IC_CHANNEL_STATE_ACTIVE \
                                            | HAL_TIM_ACTIVE_SILENT),
  #endif /* USE_HAL_TIM_DMA */

} hal_tim_channel_state_t;

Note

HAL2 provides a more detailed enumeration, distinguishing between output and input channel states, and includes additional states for DMA silent mode.

  • uint32_t BreakState and uint32_t Break2State are configured using the function HAL_TIM_BREAK_EnableInput .

  • uint32_t AutomaticOutput is configured using the function HAL_TIM_BREAK_EnableAutomaticOutput .
