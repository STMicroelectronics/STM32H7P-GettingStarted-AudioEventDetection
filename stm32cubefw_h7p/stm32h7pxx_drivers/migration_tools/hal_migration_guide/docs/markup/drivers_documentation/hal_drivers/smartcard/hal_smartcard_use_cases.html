HAL SMARTCARD Typical use cases for migration ¶

This section provides information about the HAL SMARTCARD Typical use cases for migration.

Note

The processes for transmitting and receiving data using polling, interrupt, or DMA remain unchanged in HAL1 and HAL2.

Note

In both HAL1 and HAL2, the size parameter in the transmit and receive APIs represents the amount of data in bytes.

#define DATA_TO_SEND_SIZE           256
#define DATA_TO_RECEIVE_SIZE       DATA_TO_SEND_SIZE

uint8_t aTxBuffer[] = " **** Smartcard Communication **** ";
uint8_t aRxBuffer[DATA_TO_RECEIVE_SIZE];

SMARTCARD_HandleTypeDef hsmartcard;


HAL_SMARTCARD_Init(&hsmartcard);




HAL_SMARTCARD_Receive(&hsmartcard, aRxBuffer,
                      DATA_TO_RECEIVE_SIZE, 1000);

HAL_SMARTCARD_Transmit(&hsmartcard, aTxBuffer,
                       DATA_TO_SEND_SIZE, 1000);

#define DATA_TO_SEND_SIZE           256
#define DATA_TO_RECEIVE_SIZE       DATA_TO_SEND_SIZE

uint8_t aTxBuffer[] = " **** Smartcard Communication **** ";
uint8_t aRxBuffer[DATA_TO_RECEIVE_SIZE];

hal_smartcard_handle_t hsmartcard;
hal_smartcard_config_t smartcard_config;

HAL_SMARTCARD_Init(&hsmartcard, HAL_SMARTCARD4);

/* add config for smartcard instance */
HAL_SMARTCARD_SetConfig(&hsmartcard, &smartcard_config);

HAL_SMARTCARD_Receive(&hsmartcard, aRxBuffer,
                      DATA_TO_RECEIVE_SIZE, 1000);

HAL_SMARTCARD_Transmit(&hsmartcard, aTxBuffer,
                       DATA_TO_SEND_SIZE, 1000);
