 int main(void)
 {
  RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};
  RCC_OscInitTypeDef RCC_OscInitStruct = {0};

  /* -1- Select MSI as system clock source to allow modification of the PLL configuration */
  RCC_ClkInitStruct.ClockType       = RCC_CLOCKTYPE_SYSCLK;
  RCC_ClkInitStruct.SYSCLKSource    = RCC_SYSCLKSOURCE_MSI;
  if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_0) != HAL_OK)
  {
    /* Initialization Error */
    Error_Handler();
  }

  /* -2- Enable HSI Oscillator, select it as PLL source and finally activate the PLL */
  RCC_OscInitStruct.OscillatorType       = RCC_OSCILLATORTYPE_HSI;
  RCC_OscInitStruct.HSIState             = RCC_HSI_ON;
  RCC_OscInitStruct.HSICalibrationValue  = RCC_HSICALIBRATION_DEFAULT;
  RCC_OscInitStruct.PLL.PLLState         = RCC_PLL_ON;
  RCC_OscInitStruct.PLL.PLLSource        = RCC_PLLSOURCE_HSI;
  RCC_OscInitStruct.PLL.PLLM             = 2;
  RCC_OscInitStruct.PLL.PLLN             = 40;
  RCC_OscInitStruct.PLL.PLLP             = 2;
  RCC_OscInitStruct.PLL.PLLQ             = 2;
  RCC_OscInitStruct.PLL.PLLR             = 4;

  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
  {
    /* Initialization Error */
    Error_Handler();
  }

  /* -3- Select the PLL as system clock source and configure the bus clocks dividers */
  RCC_ClkInitStruct.ClockType       = (RCC_CLOCKTYPE_SYSCLK | RCC_CLOCKTYPE_HCLK
                                       | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2
                                       | RCC_CLOCKTYPE_PCLK3);
  RCC_ClkInitStruct.SYSCLKSource    = RCC_SYSCLKSOURCE_PLLCLK;
  RCC_ClkInitStruct.AHBCLKDivider   = RCC_SYSCLK_DIV1;
  RCC_ClkInitStruct.APB1CLKDivider  = RCC_HCLK_DIV1;
  RCC_ClkInitStruct.APB2CLKDivider  = RCC_HCLK_DIV1;
  RCC_ClkInitStruct.APB3CLKDivider  = RCC_HCLK_DIV1;
  if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_4) != HAL_OK)
  {
    /* Initialization Error */
    Error_Handler();
  }

  /* -4- Optional: Disable MSI Oscillator */
  RCC_OscInitStruct.OscillatorType  = RCC_OSCILLATORTYPE_MSI;
  RCC_OscInitStruct.MSIState        = RCC_MSI_OFF;
  RCC_OscInitStruct.PLL.PLLState    = RCC_PLL_NONE;  /* No update on PLL */
  if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
  {
    /* Initialization Error */
    Error_Handler();
  }

   HAL_RCC_SetBusClockConfig(&config_bus);

{

   /* -1- Select HSI as system clock source */
   if (HAL_RCC_SetSysClkSource(HAL_RCC_SYSCLKSOURCE_HSI) != HAL_OK)

     /* Initialization Error */
     Error_Handler();

   HAL_FLASH_SetLatency(flash1, FLASH_LATENCY_0);

   HAL_RCC_PLL1_Disable();

    config_bus.ahb_clk_divider  = HAL_RCC_HCLK_PRESCALER1;
    config_bus.apb1_clk_divider = HAL_RCC_PCLK_PRESCALER1;
    config_bus.apb2_clk_divider = HAL_RCC_PCLK_PRESCALER1;
    config_bus.apb3_clk_divider = HAL_RCC_PCLK_PRESCALER1;

   HAL_RCC_SetBusClockConfig(&config_bus);

   /* -3- Select the MSIS as system clock source */
   HAL_FLASH_SET_LATENCY(flash1, FLASH_LATENCY_2);
   if (HAL_RCC_SetSysClkSource(HAL_RCC_SYSCLKSOURCE_MSIS) != HAL_OK)

     /* Initialization Error */
     Error_Handler();

   if (HAL_RCC_HSI_Disable() != HAL_OK)
   {
     /* Initialization Error */
     Error_Handler();
   }
}

Note

Note: In HAL1, the configuration of the number of wait states for the flash memory was included in the RCC configuration functions. However, in HAL2, this configuration has been removed from the RCC functions and must now be done using a specific function in the flash driver.
