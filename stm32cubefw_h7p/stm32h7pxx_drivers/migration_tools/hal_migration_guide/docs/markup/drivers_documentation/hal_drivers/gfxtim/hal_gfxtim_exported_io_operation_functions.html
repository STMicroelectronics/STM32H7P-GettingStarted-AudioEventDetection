HAL GFXTIM I/O operations migration ¶

This section provides information about the HAL GFXTIM function migration.

HAL GFXTIM configure the clock generator ¶

HAL1

HAL2 

HAL_StatusTypeDef HAL_GFXTIM_ClockGenerator_Config(GFXTIM_HandleTypeDef *hgfxtim,
                                                   const GFXTIM_ClockGeneratorConfigTypeDef *pClockGeneratorConfig);

hal_status_t HAL_GFXTIM_CLKGEN_SetConfigLineFrameClock(hal_gfxtim_handle_t *hgfxtim,
                                                       const hal_gfxtim_clkgen_frame_config_t *p_frame_clk_config,
                                                       const hal_gfxtim_clkgen_line_config_t *p_line_clk_config);
hal_status_t HAL_GFXTIM_CLKGEN_SetConfigFrameClock(hal_gfxtim_handle_t *hgfxtim,
                                                   const hal_gfxtim_clkgen_frame_config_t *p_clk_config);

Usage example: 

/* Configure the GFXTIM line frame clock generator */
GFXTIM_ClockGeneratorConfigTypeDef ClockGenConfig = {
.LCCHwReloadSrc  = GFXTIM_LCC_HW_RELOAD_SRC_NONE;
.LCCReloadValue  = 159999;
.LCCClockSrc     = GFXTIM_LCC_CLK_SRC_SYSCLOCK;
.LineClockSrc    = GFXTIM_LINE_CLK_SRC_LCC_UNDERFLOW;


.FCCHwReloadSrc  = GFXTIM_FCC_HW_RELOAD_SRC_NONE;
.FCCReloadValue  = 0xFU;
.FCCClockSrc     = GFXTIM_FCC_CLK_SRC_DISABLE;
.FrameClockSrc   = GFXTIM_FRAME_CLK_SRC_FCC_UNDERFLOW;
.LineClockCalib  = GFXTIM_LINE_CLK_CALIB_DISABLE;
.FrameClockCalib = GFXTIM_FRAME_CLK_CALIB_DISABLE;
};
HAL_GFXTIM_ClockGenerator_Config(&hgfxtim,
                                 &pClockGeneratorConfig);


/* Configure the GFXTIM frame clock generator */
GFXTIM_ClockGeneratorConfigTypeDef pClockGenConfig = {
.LineClockSrc    = GFXTIM_LINE_CLK_SRC_FCC_UNDERFLOW;
.FCCHwReloadSrc  = GFXTIM_FCC_HW_RELOAD_SRC_NONE;
.FCCReloadValue  = 1;
.FCCClockSrc     = GFXTIM_FCC_CLK_SRC_HSYNC_RISING;
.FrameClockSrc   = GFXTIM_FRAME_CLK_SRC_FCC_UNDERFLOW;
.LineClockCalib  = GFXTIM_LINE_CLK_CALIB_DISABLE;
.FrameClockCalib = GFXTIM_FRAME_CLK_CALIB_DISABLE;
};
HAL_GFXTIM_ClockGenerator_Config(&hgfxtim,
                                 &pClockGenConfig);

Usage example: 

/* Configure the GFXTIM line frame clock generator */
hal_gfxtim_clkgen_line_config_t p_line_config = {
.reload_src   = HAL_GFXTIM_CLKGEN_LCNT_NO_RELOAD;
.reload_val   = 159999;
.counter_src  = HAL_GFXTIM_CLKGEN_LINE_COUNT_GFXTIMHCLK;
.clock_src    = HAL_GFXTIM_CLKGEN_LINE_LCNT_UNDERFLOW;
 };
hal_gfxtim_clkgen_frame_config_t p_frame_config = {
.reload_src  = HAL_GFXTIM_CLKGEN_FCNT_NO_RELOAD;
.reload_val  = 0xFU;
.counter_src = HAL_GFXTIM_CLKGEN_FCNT_DISABLE;
.clock_src   = HAL_GFXTIM_CLKGEN_FRAME_FCNT_UNDERFLOW;
};


HAL_GFXTIM_CLKGEN_SetConfigLineFrameClock(&hgfxtim,
                                          &p_frame_config,
                                          &p_line_config);


/* Configure the GFXTIM frame clock generator */
hal_gfxtim_clkgen_frame_config_t p_frame_config = {

.reload_src  = HAL_GFXTIM_CLKGEN_FCNT_NO_RELOAD;
.reload_val  = 1;
.counter_src = HAL_GFXTIM_CLKGEN_FCNT_HSYNC_RISING;
.clock_src   = HAL_GFXTIM_CLKGEN_FRAME_FCNT_UNDERFLOW;
};


HAL_GFXTIM_CLKGEN_SetConfigFrameClock(&hgfxtim,
                                      &p_frame_config);

HAL GFXTIM get the current counter value of the absolute timer ¶

HAL1

HAL2 

HAL_StatusTypeDef HAL_GFXTIM_AbsoluteTimer_GetCounter(const GFXTIM_HandleTypeDef *hgfxtim,
                                                      uint32_t AbsoluteTime,
                                                      uint32_t *pValue);

void HAL_GFXTIM_ABSTIM_GetCounter(hal_gfxtim_handle_t *hgfxtim,
                                  uint32_t *p_frame,
                                  uint32_t *p_line);
uint32_t HAL_GFXTIM_ABSTIM_GetFrameCounter(hal_gfxtim_handle_t *hgfxtim);
uint32_t HAL_GFXTIM_ABSTIM_GetLineCounter(hal_gfxtim_handle_t *hgfxtim);

Usage example: 

uint32_t value
/* Get the absolute line and frame counter */
HAL_GFXTIM_AbsoluteTimer_GetCounter(&hgfxmmu, GFXTIM_ABSOLUTE_GLOBAL_TIME, &value);
/* Get the absolute line counter */
HAL_GFXTIM_AbsoluteTimer_GetCounter(&hgfxmmu, GFXTIM_ABSOLUTE_LINE_TIME, &value);
/* Get the absolute frame counter */
HAL_GFXTIM_AbsoluteTimer_GetCounter(&hgfxmmu, GFXTIM_ABSOLUTE_FRAME_TIME, &value);

Usage example: 

uint32_t frame_value, line_value ;
/* Get the absolute line and frame counter */
HAL_GFXTIM_ABSTIM_GetCounter(&hgfxmmu,&frame_value, &line_value);
/* Get the absolute line counter */
line_value = HAL_GFXTIM_ABSTIM_GetLineCounter(&hgfxtim);
/* Get the absolute frame counter */
frame_value = HAL_GFXTIM_ABSTIM_GetFrameCounter(&hgfxtim);

                                                  const GFXTIM_RelativeTimerConfigTypeDef *pRelativeTimerConfig,

Removed:

The function HAL_GFXTIM_RelativeTimer_Config has been merged with HAL_GFXTIM_RelativeTimer_Start to provide starting
relative counter with the appropriate mode (continuous or one shot).

For more details, refer to HAL GFXTIM start the relative timer

HAL GFXTIM start the relative timer ¶

HAL1

HAL2 

HAL_StatusTypeDef HAL_GFXTIM_RelativeTimer_Start(GFXTIM_HandleTypeDef *hgfxtim,
                                                 uint32_t RelativeTimer);

hal_status_t HAL_GFXTIM_RELTIM_StartContinuous(hal_gfxtim_handle_t *hgfxtim,
                                               hal_gfxtim_reltim_counter_t counter,
                                               uint32_t autoreload_val,
                                               uint32_t interrupt);

hal_status_t HAL_GFXTIM_RELTIM_StartOneShot(hal_gfxtim_handle_t *hgfxtim,
                                            hal_gfxtim_reltim_counter_t counter,
                                            uint32_t autoreload_val,
                                            uint32_t interrupt);

Usage example: 

/* Configure and start the relative frame counter in one shot mode */
GFXTIM_RelativeTimerConfigTypeDef pRelativeTimerConfig = {
.CounterMode = GFXTIM_MODE_ONE_SHOT;

.AutoReloadValue = 200;
.ReloadInterrupt = ENABLE;
};
HAL_GFXTIM_RelativeTimer_Config(&hgfxtim,
                                &pRelativeTimerConfig,
                                GFXTIM_RELATIVE_TIMER1);
HAL_GFXTIM_RelativeTimer_Start(&hgfxtim,
                                GFXTIM_RELATIVE_TIMER1);

/* Configure and start the relative frame counter in continuous mode */
GFXTIM_RelativeTimerConfigTypeDef pRelativeTimerConfig = {
.CounterMode = GFXTIM_MODE_CONTINUOUS;

.AutoReloadValue = 200;
.ReloadInterrupt = ENABLE;
};
HAL_GFXTIM_RelativeTimer_Config(&hgfxtim,
                                &pRelativeTimerConfig,
                                GFXTIM_RELATIVE_TIMER2);
HAL_GFXTIM_RelativeTimer_Start(&hgfxtim,
                                GFXTIM_RELATIVE_TIMER2);

Usage example: 

/* Configure and start the relative frame counter in one shot mode */

HAL_GFXTIM_RELTIM_StartOneShot(&hgfxtim,
                               HAL_GFXTIM_RELTIM_COUNTER_1,
                               200U,
                               HAL_GFXTIM_RELTIM_COUNT_IT);







/* Configure and start the relative frame counter in continuous mode */

HAL_GFXTIM_RELTIM_StartContinuous(&hgfxtim,
                                HAL_GFXTIM_RELTIM_COUNTER_2,
                                200U,
                                HAL_GFXTIM_RELTIM_COUNT_IT);
