HAL PWR Enter Stop Mode with RTC Wakeup ¶

// RTC handle
RTC_HandleTypeDef hrtc;

void SystemClock_Config(void)
{
    // Reinitialize the system clock after waking up
    // (Implementation depends on your specific clock configuration)
    // Example: Re-enable HSE or HSI and configure PLL

{
    // HAL initialization
    HAL_Init();

    // Configure the system clock
    SystemClock_Config();

    // Initialize RTC
    hrtc.Instance = RTC;
    HAL_RTC_Init(&hrtc);

    // Enable PWR clock
    __HAL_RCC_PWR_CLK_ENABLE();

    // Configure Voltage Scaling (Scale 1 for low-power modes)
    HAL_PWREx_ControlVoltageScaling(PWR_REGULATOR_VOLTAGE_SCALE1);

    // Configure the RTC as a wakeup source
    HAL_RTCEx_SetWakeUpTimer_IT(&hrtc, 0xFFFF, RTC_WAKEUPCLOCK_RTCCLK_DIV16 , 0);

    // Enter Stop Mode (Stop 0 with low-power regulator)
    HAL_PWR_EnterSTOPMode(PWR_LOWPOWERREGULATOR_ON, PWR_STOPENTRY_WFI);

    // System resumes here after wakeup
    SystemClock_Config(); // Reconfigure the system clock

    while (1)
    {
        // Main loop
    }
}

#include "stm32u5xx_hal.h"

// Function Prototypes
void SystemClock_Config(void);

// Main Function
int main(void)
{
    // Initialize the HAL Library
    HAL_Init();

    // Configure the System Clock
    SystemClock_Config();

    // Configure the Power Voltage Detector (PVD)
    PWR_PVDTypeDef sConfigPVD;

    // Enable the PWR Clock
    HAL_RCC_PWR_EnableClock();

    // Configure the PVD Level and Mode
    sConfigPVD.PVDLevel = PWR_PVDLEVEL_3; // Set threshold (e.g., 2.5V)
    sConfigPVD.Mode = PWR_PVD_MODE_IT_RISING_FALLING; // Interrupt on both rising and falling edges

    // Initialize the PVD
    HAL_PWR_ConfigPVD(&sConfigPVD);

    // Enable the PVD Interrupt
    HAL_PWR_EnablePVD();

    // Configure the EXTI Line for PVD
    HAL_NVIC_SetPriority(PVD_AVD_IRQn, 0, 0); // Set priority
    HAL_NVIC_EnableIRQ(PVD_AVD_IRQn);         // Enable the interrupt

    // Enter Sleep Mode and wait for PVD interrupt
    HAL_PWR_EnterSLEEPMode(PWR_MAINREGULATOR_ON, PWR_SLEEPENTRY_WFI);

    // Main loop (should not reach here unless woken up)
    while (1)
    {
        // Add your application code here
    }
}

Usage 

// Function prototypes
void SystemClock_Config(void);

int main(void)
{
    // HAL initialization
    HAL_Init();
    SystemClock_Config();

    // Initialize GPIO for EXTI
    GPIO_InitTypeDef GPIO_InitStruct = {0};

    // Enable GPIOA clock
    __HAL_RCC_GPIOA_CLK_ENABLE();

    // Configure PA0 as input with EXTI interrupt
    GPIO_InitStruct.Pin = GPIO_PIN_0;
    GPIO_InitStruct.Mode = GPIO_MODE_IT_RISING; // Trigger on rising edge
    GPIO_InitStruct.Pull = GPIO_NOPULL;        // No pull-up or pull-down
    HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);

    // Enable and set EXTI line 0 interrupt priority
    HAL_NVIC_SetPriority(EXTI0_IRQn, 0, 0);
    HAL_NVIC_EnableIRQ(EXTI0_IRQn);

    // Enter standby mode
    // Enable Power Control clock
    __HAL_RCC_PWR_CLK_ENABLE();

    // Clear the Wakeup flag
    __HAL_PWR_CLEAR_FLAG(PWR_FLAG_WU);

    // Enable EXTI wakeup source
    HAL_PWR_EnableWakeUpPin(PWR_WAKEUP_PIN1_HIGH);

    // Enter Standby mode
    HAL_PWR_EnterSTANDBYMode();

    // Code should never reach here after entering standby mode
    while (1)
    {
    }
}
