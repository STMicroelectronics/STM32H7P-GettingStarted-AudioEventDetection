HAL ICACHE remap region successful use case ¶

This use case demonstrates how to enable the remapping region with ICACHE functions using HAL1 compared to HAL2.

We have an extra function HAL_ICACHE_SetConfigRemapRegion() to configure the handle, the region and the configuration struct.

Then we can enable the remapping of our region and start the process.

HAL1 Example

HAL2 Example 

HAL_ICACHE_Disable();

HAL_ICACHE_ConfigAssociativityMode(ICACHE_1WAY);

/* Remap SRAM memory */
region_config.TrafficRoute    = ICACHE_MASTER2_PORT;
region_config.OutputBurstType = ICACHE_OUTPUT_BURST_INCR;
region_config.Size            = ICACHE_REGIONSIZE_2MB;
region_config.BaseAddress     = 0x10000000;
region_config.RemapAddress    = 0x60000000;

/* Enable the remapped region */
if(HAL_ICACHE_EnableRemapRegion(ICACHE_REGION_1, &region_config) != HAL_OK)
{
  Error_Handler();
}

/* Enable the Instruction and Data Cache */
HAL_ICACHE_Enable();

HAL_ICACHE_Init(&hicache1, HAL_ICACHE1);

HAL_ICACHE_SetAssociativityMode(&hicache1, HAL_ICACHE_ASSOCIATIVITY_1WAY);

/* Remap SRAM memory */
region_config.master_port   = HAL_ICACHE_MASTER2_PORT;
region_config.output_burst  = HAL_ICACHE_BURST_INCR;
region_config.size          = HAL_ICACHE_REGION_SIZE_2MBYTES;
region_config.base_address  = 0x10000000;
region_config.remap_address = 0x60000000;

/* Enable the remapped region */
if(HAL_ICACHE_SetConfigRemapRegion(&hicache1, HAL_ICACHE_REGION_1, &region_config) != HAL_OK)
{
  Error_Handler();
}

if(HAL_ICACHE_EnableRemapRegion(&hicache1, HAL_ICACHE_REGION_1) != HAL_OK)
{
  Error_Handler();
}

/* Start the Instruction Cache */
HAL_ICACHE_Start(&hicache1, HAL_ICACHE_IT_NONE);
